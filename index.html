function updateClock() { const now = new Date(); const mountainTime = new Intl.DateTimeFormat("en-US", { timeZone: "America/Denver", hour: "2-digit", minute: "2-digit", hour12: true }); document.getElementById("day").textContent = new Intl.DateTimeFormat("en-US", { timeZone: "America/Denver", weekday: "long" }).format(now); document.getElementById("date").textContent = new Intl.DateTimeFormat("en-US", { timeZone: "America/Denver", month: "long", day: "numeric", year: "numeric" }).format(now); document.getElementById("clock").textContent = mountainTime.format(now); } updateClock(); setInterval(updateClock, 1000);
